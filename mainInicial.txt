package main

import (
	"fmt"
	"log"
	"net/http"
)

func main() {

	fs := http.FileServer(http.Dir("./static")) //Aqui definimos a pasta "static" como raiz dos arquivos públicos

	http.Handle("/", fs) //Toda requisição "/" servirá os arquivos da pasta static

	porta := ":8080" //Porta que o servidor irá rodar

	fmt.Println("Servidor rodando em http://localhost" + porta)
	fmt.Println("	Pressione Ctrl+C para parar.")

	err := http.ListenAndServe(porta, nil) //Inicia o servidor, caso dê erro, loga e para.
	if err != nil {
		log.Fatal("Erro ao iniciar o servidor: ", err)
	}
}
